기존 CSV에서 특정 index 구간을 INTERSECTION으로 변경
cd ~/mmission_ws

python3 - <<'PY'
import csv

path = "/home/qor/mmission_ws/routes/full_course.csv"
start_index = 100
end_index = 180

with open(path, newline="", encoding="utf-8") as f:
    rows = list(csv.DictReader(f))
    fields = rows[0].keys()

for row in rows:
    i = int(row["index"])
    if start_index <= i <= end_index:
        row["mode"] = "INTERSECTION"

with open(path, "w", newline="", encoding="utf-8") as f:
    w = csv.DictWriter(f, fieldnames=fields)
    w.writeheader()
    w.writerows(rows)
PY
